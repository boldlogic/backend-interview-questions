type Getter struct {
	url string
}

type WrapGetter struct {
	g *Getter
}

func NewGetter(url string) *WrapGetter {
	return &WrapGetter{
		g: &Getter{
			url: url,
		},
	}
}

func (g *WrapGetter) GetPrice() (float64, error) {
	return g.g.GetPrice()
}
