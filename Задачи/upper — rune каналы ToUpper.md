```go
package main

import (
	"fmt"
	"unicode"
)

// В этом задании вам нужно реализовать функцию upper().
// Она должна принимать в параметрах два канала.
// Первый — для получения символов типа rune, второй — для записи struct{}, чтобы известить программу о завершении работы.
// Сделайте эти каналы однонаправленными.
// Все прочитанные символы следует сделать заглавными и вывести на консоль без дополнительных переводов строк.
// Когда канал для чтения закроется, следует отправить пустую структуру во второй канал.
//
//	В функции main() требуется только одна доработка — создайте буферизированный канал c рунами ёмкостью 1.
func upper(in <-chan rune, done chan<- struct{}) {
	// напишите код функции
loop:
	for {
		select {
		case v, ok := <-in:
			if !ok {
				done <- struct{}{}
				break loop
			}
			fmt.Printf("%c", unicode.ToUpper(v))
		}
	}

}

func main() {
	// определите буферизированный канал с рунами (rune)
	ch := make(chan rune) // ...
	done := make(chan struct{})

	go upper(ch, done)
	for _, v := range "Каждый охотник желает знать, где сидит фазан" {
		ch <- v
	}
	close(ch)
	<-done
}
```
