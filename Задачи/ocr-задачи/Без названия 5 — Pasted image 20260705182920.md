func (g *WrapGetter) GetPrice() (float64, error) {
	g.mu.Lock()
	i:=g.idx%len(g.backends)
	g.idx++
	g.mu.Unlock()

	res,err:=g.backends[i]
	if err!=nil{
		g.mu.Lock
		g.backends=append(g.backends[:i],g.backends[i+1:])
		g.bans [i]=1
		g.mu.Unlock()
	}
	return
}
func (g *WrapGetter) Cron(ttl time.Duration){
	tick:=time.NewTicker
	for {
		<-tick.C
		for i:=range g.bans{
			if bans[i]==1{
				g.mu.Lock()
				g.backends=append(g.bans[i])
				g.mu.Unlokk()
			}
		}
	}
}
